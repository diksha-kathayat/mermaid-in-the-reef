    <script>

        const canvas = document.getElementById('gameCanvas');

        const ctx = canvas.getContext('2d');

        const scoreUI = document.getElementById('scoreUI');

        const msgContainer = document.getElementById('msgContainer');


        // --- MOBILE LAYOUT ---

        function resize() {

            canvas.width = window.innerWidth;

            canvas.height = window.innerHeight;

        }

        window.addEventListener('resize', resize);

        resize();


        let mermaid = { x: 70, y: 0, v: 0, gravity: 0.25, jump: -6, rotation: 0 };

        let pipes = [], bubbles = [], noseBubbles = [], wildlife = [], stones = [];

        let score = 0, gameActive = false, dead = false, frame = 0, saturation = 100;


        // --- AUDIO ---

        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

        function playSfx(f, t, d, v) {

            try {

                const o = audioCtx.createOscillator(); const g = audioCtx.createGain();

                o.type = t; o.frequency.setValueAtTime(f, audioCtx.currentTime);

                g.gain.setValueAtTime(v, audioCtx.currentTime);

                g.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + d);

                o.connect(g); g.connect(audioCtx.destination); o.start(); o.stop(audioCtx.currentTime + d);

            } catch(e) {}

        }


        // --- COMPLEX DRAWING FUNCTIONS ---

        // Draws cascading, heavily wavy hair using Bézier curves and dynamic offsets

        function drawHeavilyWavyHair(x, y, op, layerOffset, isFront = false) {

            ctx.fillStyle = `hsla(158, 70%, 80%, ${op})`; // Mint Green Hair

            ctx.beginPath();

            

            // Generate multiple cascading strands

            for (let strand = 0; strand < 4; strand++) {

                let yStart = y - 12 + (strand * 5);

                // Dynamically offset the hair's movement based on the frame and layer

                let motionOffset = Math.sin((frame + layerOffset + strand*15) * 0.12) * 5;

                let strandLength = 55 + strand * 5;

                

                // Hair root/start

                ctx.moveTo(x + (isFront ? 8 : -5), yStart);

                

                // Dynamic flow back (undulation) using a double Bezier curve

                // ctx.bezierCurveTo(CP1x, CP1y, CP2x, CP2y, Ex, Ey)

                ctx.bezierCurveTo(

                    x - (isFront ? 10 : 20), yStart - 5 + motionOffset, // CP1

                    x - 30, yStart + 20 + motionOffset, // CP2

                    x - strandLength, yStart + 5 + motionOffset // End point

                );

                

                // Curl and flow back to root

                ctx.bezierCurveTo(

                    x - 45, yStart + 25 + motionOffset, 

                    x - 30, yStart + 5 + motionOffset, 

                    x + (isFront ? 8 : -5), yStart // Back to root

                );

            }

            ctx.fill();

        }


        function drawWildlife(w) {

            ctx.fillStyle = `rgba(255,255,255,${w.op})`;

            ctx.beginPath();

            if(w.type === 'shark') {

                ctx.ellipse(w.x, w.y, 45, 12, 0, 0, Math.PI*2); // Body

                ctx.moveTo(w.x-5, w.y-10); ctx.lineTo(w.x-20, w.y-30); ctx.lineTo(w.x+5, w.y-8); // Top Fin

                ctx.moveTo(w.x-40, w.y); ctx.lineTo(w.x-60, w.y-20); ctx.lineTo(w.x-52, w.y); ctx.lineTo(w.x-60, w.y+20); // Tail

            } else if(w.type === 'turtle') {

                ctx.ellipse(w.x, w.y, 22, 18, 0, 0, Math.PI*2); // Shell

                ctx.arc(w.x+28, w.y, 8, 0, Math.PI*2); // Head

                ctx.ellipse(w.x+8, w.y-18, 14, 6, -Math.PI/4, 0, Math.PI*2); // Front Fin

                ctx.ellipse(w.x+8, w.y+18, 14, 6, Math.PI/4, 0, Math.PI*2); // Front Fin

            } else {

                ctx.ellipse(w.x, w.y, 14, 9, 0, 0, Math.PI*2);

                ctx.moveTo(w.x-10, w.y); ctx.lineTo(w.x-22, w.y-10); ctx.lineTo(w.x-18, w.y); ctx.lineTo(w.x-22, w.y+10);

            }

            ctx.fill();

        }


        // --- CORE CORE ---

        function initStones() {

            stones = [];

            for(let i=0; i<12; i++) stones.push({ x: Math.random()*canvas.width, y: canvas.height-35-Math.random()*15, r: Math.random()*12+6 });

        }


        function reset() {

            mermaid.y = canvas.height / 2; mermaid.v = 0;

            pipes = []; bubbles = []; noseBubbles = []; wildlife = [];

            score = 0; frame = 0; saturation = 100; dead = false; gameActive = true;

            scoreUI.innerText = "0"; msgContainer.style.display = "none";

            initStones();

            if (audioCtx.state === 'suspended') audioCtx.resume();

            playSfx(440, 'sine', 0.2, 0.05);

            // Pre-fill background bubbles

            for(let i=0; i<30; i++) bubbles.push({x: Math.random()*canvas.width, y: Math.random()*canvas.height, s: Math.random()*1.5+0.5, r: Math.random()*4+1});

        }


        // Emit bubbles from her nose when she swims

        function emitNoseBubbles() {

            // Emits 3-5 bubbles

            for(let i=0; i<4; i++) {

                noseBubbles.push({

                    x: mermaid.x + 20, // Nose location

                    y: mermaid.y - 12, // Adjusted for head position

                    vx: -3 - Math.random() * 2, vy: -1 - Math.random() * 2, // Drift backwards and upwards

                    r: Math.random() * 2 + 1.5,

                    life: 1 // For fade out

                });

            }

        }


        function update() {

            frame++;

            if (dead) { if (saturation > 30) saturation -= 1.5; return; }

            if (!gameActive) return;


            mermaid.v += mermaid.gravity;

            mermaid.y += mermaid.v;

            // Familiar Flappy Physics rotation

            mermaid.rotation = Math.min(Math.PI/4, Math.max(-Math.PI/6, mermaid.v * 0.08));


            // Background Bubbles

            if (frame % 5 === 0) bubbles.push({x: Math.random()*canvas.width, y: canvas.height+20, s: Math.random()*2+1, r: Math.random()*4+1.5});

            bubbles.forEach(b => b.y -= b.s);

            bubbles = bubbles.filter(b => b.y > -20);

            

            // Nose Bubbles

            noseBubbles.forEach(nb => {

                nb.x += nb.vx; nb.y += nb.vy;

                nb.vy -= 0.1; // Float acceleration

                nb.vx *= 0.98; // Friction

                nb.life -= 0.02; // Fade out

            });

            noseBubbles = noseBubbles.filter(nb => nb.y > -20 && nb.life > 0);


            if (frame % 280 === 0) wildlife.push({

                x: canvas.width + 100, y: Math.random()*(canvas.height-200)+100, 

                type: ['shark','turtle','fish'][Math.floor(Math.random()*3)], 

                speed: Math.random()*1.2+0.6, op: 0.15

            });

            wildlife.forEach(w => w.x -= w.speed);


            if (frame % 105 === 0) {

                let h = Math.random() * (canvas.height - 400) + 100;

                pipes.push({ x: canvas.width, top: h, gap: 195, passed: false });

            }

            pipes.forEach(p => {

                p.x -= 3.3;

                if (mermaid.x + 35 > p.x && mermaid.x - 35 < p.x + 65) {

                    if (mermaid.y - 18 < p.top || mermaid.y + 18 > p.top + p.gap) {

                        dead = true; gameActive = false;

                        playSfx(100, 'sawtooth', 0.5, 0.1);

                        msgContainer.style.display = "block";

                    }

                }

                if (!p.passed && p.x < mermaid.x) { score++; p.passed = true; scoreUI.innerText = score; playSfx(900, 'sine', 0.2, 0.03); }

            });

            if (mermaid.y > canvas.height - 85 || mermaid.y < 0) { dead = true; gameActive = false; msgContainer.style.display = "block"; }

        }


        function draw() {

            ctx.filter = `saturate(${saturation}%)`;

            let grad = ctx.createLinearGradient(0,0,0,canvas.height);

            grad.addColorStop(0, "#014f86"); grad.addColorStop(1, "#4cc9f0");

            ctx.fillStyle = grad; ctx.fillRect(0,0,canvas.width, canvas.height);


            // BUBBLES (Background)

            ctx.fillStyle="rgba(255,255,255,0.22)";

            bubbles.forEach(b => { ctx.beginPath(); ctx.arc(b.x, b.y, b.r, 0, Math.PI*2); ctx.fill(); });

            

            wildlife.forEach(w => drawWildlife(w));


            // Pipes (Coral)

            pipes.forEach(p => {

                ctx.fillStyle = "#ff8a80"; ctx.strokeStyle = "white"; ctx.lineWidth = 4;

                ctx.beginPath(); ctx.roundRect(p.x, 0, 65, p.top, [0,0,15,15]); ctx.fill(); ctx.stroke();

                ctx.beginPath(); ctx.roundRect(p.x, p.top + p.gap, 65, canvas.height, [15,15,0,0]); ctx.fill(); ctx.stroke();

            });


            // Sand & Stones

            ctx.fillStyle = "#f3e5ab"; ctx.beginPath(); ctx.moveTo(0, canvas.height);

            for(let i=0; i<=canvas.width; i+=10) {

                let wave = Math.sin((i + frame) * 0.02) * 6;

                ctx.lineTo(i, canvas.height - 85 + wave);

            }

            ctx.lineTo(canvas.width, canvas.height); ctx.fill();


            for (let i = 0; i < canvas.width; i += 25) {

                ctx.fillStyle = "rgba(0,0,0,0.05)"; ctx.fillRect(i + Math.sin(i)*10, canvas.height - 40, 2, 2);

            }

            stones.forEach(s => { ctx.fillStyle = "#90a4ae"; ctx.beginPath(); ctx.ellipse(s.x, s.y, s.r, s.r/1.5, Math.PI/4, 0, Math.PI*2); ctx.fill(); });


            if (!dead) {

                ctx.save(); ctx.translate(mermaid.x, mermaid.y); ctx.rotate(mermaid.rotation);

                

                // --- GIRL MERMAID AESTHETIC ---

                

                // Hair Layer 1 (Back Wavy Hair)

                drawHeavilyWavyHair(0, -10, 1, 0);


                // Body & CLASSIC FISH TAIL (Gradient Pink -> Teal)

                let tG = ctx.createLinearGradient(-10, 0, -50, 0); 

                tG.addColorStop(0, "#fbcfe8"); // Pink transition

                tG.addColorStop(1, "#2dd4bf"); // Teal main

                ctx.fillStyle = tG; 

                ctx.beginPath(); 

                ctx.ellipse(-15, 0, 32, 10, 0, 0, Math.PI*2); // Tapered Body/Tail shape

                ctx.fill(); 

                

                // Classic Detailed Fin (Pink)

                ctx.fillStyle = "#f472b6"; 

                ctx.beginPath(); 

                ctx.ellipse(-50, 0, 18, 12, Math.PI/2, 0, Math.PI*2); // Fin shape

                ctx.fill();


                // Head (Skin rounded Face)

                ctx.fillStyle = "#ffe0bd"; ctx.beginPath(); ctx.arc(10, -5, 11, 0, Math.PI*2); ctx.fill(); 

                

                // Hair Layer 2 (Wavy Hair Strand over face)

                drawHeavilyWavyHair(10, -12, 1, 30, true);


                // Eyes (Disney Style with Sparkle)

                ctx.fillStyle = "white"; ctx.beginPath(); ctx.arc(16, -7, 5, 0, Math.PI*2); ctx.fill(); 

                ctx.fillStyle = "black"; ctx.beginPath(); ctx.arc(18, -7, 3, 0, Math.PI*2); ctx.fill();

                ctx.fillStyle = "white"; ctx.beginPath(); ctx.arc(19, -8, 1, 0, Math.PI*2); ctx.fill(); // Sparkle


                ctx.restore();

            }


            // NOSE BUBBLES (Emitted when swimming - drawn in world space)

            ctx.fillStyle="rgba(255,255,255,0.7)";

            noseBubbles.forEach(nb => {

                ctx.beginPath(); ctx.arc(nb.x, nb.y, nb.r, 0, Math.PI*2); ctx.fill();

            });


            requestAnimationFrame(() => { update(); draw(); });

        }


        const handleInput = (e) => {

            if (e.type === 'touchstart') e.preventDefault();

            if(!gameActive) reset(); else { 

                mermaid.v = mermaid.jump; 

                emitNoseBubbles(); // Add Nose Bubbles on Swim

                playSfx(350, 'sine', 0.1, 0.03); 

            }

        };

        window.addEventListener('mousedown', handleInput);

        window.addEventListener('touchstart', handleInput, {passive: false});

        draw();

    </script>
